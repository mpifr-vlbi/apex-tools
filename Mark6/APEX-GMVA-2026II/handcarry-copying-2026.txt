## C262A
# Disk 2 : C262A pol 1
# Disk 3 : C262A pol 2

du -h -c /mark6-3_fuse/{1,2}/c262a*
  27T total
rsync -axv --progress --inplace --no-compress /mark6-3_fuse/1/c262a_Ax_* /mnt/handcarry/disk2/slot1/  # ~13 TB out of disk size 20 TB, LCP
rsync -axv --progress --inplace --no-compress /mark6-3_fuse/2/c262a_Ax_* /mnt/handcarry/disk3/slot2/  # ~13 TB out of disk size 20 TB, RCP


## C262B
# Disk 4 : C262B pol 1 and pol 2
